document.addEventListener("DOMContentLoaded", function () {
  const bar = document.querySelector("[data-selection-bar]");
  const countLabel = document.querySelector("[data-selection-count]");

  const actionModal = document.querySelector("[data-action-modal]");
  const promiseModal = document.querySelector("[data-promise-modal]");

  function getSelectedDocs() {
    return Array.from(document.querySelectorAll("[data-account-doc-checkbox]:checked"));
  }

  function parseAmount(raw) {
    if (!raw) return 0;
    return Number(String(raw).replace(",", ".")) || 0;
  }

  function formatCLP(amount) {
    return new Intl.NumberFormat("es-CL", {
      style: "currency",
      currency: "CLP",
      maximumFractionDigits: 0
    }).format(amount);
  }

  function updateSelectionBar() {
    if (!bar || !countLabel) return;

    const count = getSelectedDocs().length;

    if (count === 0) {
      bar.hidden = true;
      countLabel.textContent = "0 documentos seleccionados";
      return;
    }

    bar.hidden = false;
    countLabel.textContent = count === 1
      ? "1 documento seleccionado"
      : `${count} documentos seleccionados`;
  }

  function fillModal(modal, hiddenSelector, summarySelector) {
    const selected = getSelectedDocs();
    const hiddenBox = modal.querySelector(hiddenSelector);
    const summary = modal.querySelector(summarySelector);
    const total = modal.querySelector("[data-promise-modal-total]");

    hiddenBox.innerHTML = "";

    selected.forEach(function (checkbox) {
      const input = document.createElement("input");
      input.type = "hidden";
      input.name = modal === actionModal ? "document_id" : "document_ids";
      input.value = checkbox.value;
      hiddenBox.appendChild(input);
    });

    if (summary) {
      summary.textContent = selected.length === 1
        ? "1 documento seleccionado"
        : `${selected.length} documentos seleccionados`;
    }

    if (total && modal === promiseModal) {
      const amount = selected.reduce(function (sum, checkbox) {
        return sum + parseAmount(checkbox.dataset.documentBalanceRaw);
      }, 0);

      total.textContent = "Total comprometido: " + formatCLP(amount);
    }
  }

  function openModal(modal) {
    if (!modal || !getSelectedDocs().length) return;
    modal.hidden = false;
  }

  function closeModals() {
    document.querySelectorAll(".if-modal").forEach(function (modal) {
      modal.hidden = true;
    });
  }

  document.querySelectorAll("[data-account-doc-checkbox]").forEach(function (checkbox) {
    checkbox.addEventListener("change", updateSelectionBar);
  });

  const actionButton = document.querySelector("[data-open-action-modal]");
  if (actionButton) {
    actionButton.addEventListener("click", function () {
      fillModal(actionModal, "[data-action-hidden-documents]", "[data-action-modal-summary]");
      openModal(actionModal);
    });
  }

  const promiseButton = document.querySelector("[data-open-promise-modal]");
  if (promiseButton) {
    promiseButton.addEventListener("click", function () {
      fillModal(promiseModal, "[data-promise-hidden-documents]", "[data-promise-modal-summary]");
      openModal(promiseModal);
    });
  }

  document.querySelectorAll("[data-close-modal]").forEach(function (button) {
    button.addEventListener("click", closeModals);
  });

  updateSelectionBar();
});
